/// Formats environment variables as a Configuration table for LLM consumption.
/// Output is prefixed with "(Detected)" to signal it's auto-extracted.
fn format_env_vars(env_vars: &[String]) -> Option<String> {
    if env_vars.is_empty() {
        return None;
    }

    let mut lines = vec![
        "## Environment Variables (Detected)".to_string(),
        String::new(),
        "| Variable | Notes |".to_string(),
        "|----------|-------|".to_string(),
    ];
    for var in env_vars {
        lines.push(format!("| `{}` | required — fill in type/default in Configuration |", var));
    }
    Some(lines.join("\n"))
}

## Flows
